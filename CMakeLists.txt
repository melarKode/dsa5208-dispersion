cmake_minimum_required(VERSION 3.23)
project(dispersion VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DISPERSION_BUILD_PARALLEL "Build MPI+Kokkos targets" OFF)
option(DISPERSION_SANITIZE "Enable address and undefined behavior checks" OFF)
add_library(dispersion_core src/core.cpp src/run.cpp)
target_include_directories(dispersion_core PUBLIC include)
target_compile_options(dispersion_core PRIVATE -Wall -Wextra -Wpedantic)
if(DISPERSION_SANITIZE)
  target_compile_options(dispersion_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(dispersion_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(dispersion_serial src/serial.cpp)
target_link_libraries(dispersion_serial PRIVATE dispersion_core)
if(DISPERSION_BUILD_PARALLEL)
  find_package(MPI REQUIRED COMPONENTS CXX)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/external/kokkos/CMakeLists.txt")
    set(Kokkos_ENABLE_SERIAL ON CACHE BOOL "Serial backend")
    add_subdirectory(external/kokkos)
  else()
    find_package(Kokkos 4.5 CONFIG REQUIRED)
  endif()
  add_executable(dispersion_mpi src/parallel.cpp)
  target_link_libraries(dispersion_mpi PRIVATE dispersion_core MPI::MPI_CXX Kokkos::kokkos)
  add_executable(mpi_kokkos_smoke src/mpi_kokkos_smoke.cpp)
  target_link_libraries(mpi_kokkos_smoke PRIVATE MPI::MPI_CXX Kokkos::kokkos)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_core tests/test_core.cpp)
  target_link_libraries(test_core PRIVATE dispersion_core)
  add_test(NAME core COMMAND test_core)
endif()
